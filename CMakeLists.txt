cmake_minimum_required(VERSION 3.20)
project(LooseGrownDiamondPriceTracker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(LooseGrownDiamondPriceTracker src/main.cpp)
if(MSVC)
    target_compile_options(LooseGrownDiamondPriceTracker PRIVATE /utf-8)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(nivoda_tests tests/nivoda_tests.cpp)
    target_compile_definitions(nivoda_tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    if(MSVC)
        target_compile_options(nivoda_tests PRIVATE /utf-8)
    endif()
    add_test(NAME nivoda_sources COMMAND nivoda_tests)
    add_executable(loosegrown_tests tests/loosegrown_tests.cpp)
    if(MSVC)
        target_compile_options(loosegrown_tests PRIVATE /utf-8)
    endif()
    add_test(NAME loosegrown_inventory COMMAND loosegrown_tests)
endif()
